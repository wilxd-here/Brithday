<script>
const API_BASE = '/api/will-movies';

/* Kategori rail. Hanya yang didukung API yang aktif.
   Tambah kategori baru di sini saat API siap, mis:
   { id:'action', title:'Action', params:{ action:'genre', genre:'action' } } */
const RAILS = [
  { id: 'home',   title: 'Terbaru Ditambahkan', params: { action: 'home' } },
  { id: 'rating', title: 'Rating Tertinggi',    params: { action: 'rating' } }
];

const $ = id => document.getElementById(id);
const esc = s => String(s ?? '').replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
const store = {
  get(k){ try { return JSON.parse(localStorage.getItem(k)) || []; } catch(e){ return []; } },
  set(k,v){ try { localStorage.setItem(k, JSON.stringify(v)); } catch(e){} }
};

const detailView = $('detail-view'), detailIframe = $('detail-iframe'), loadingOverlay = $('loading-overlay');
const detailTitle = $('info-title'), detailServerGrid = $('detail-server-grid');
const movieContainer = $('movie-container'), heroBanner = $('hero-banner');
const searchInput = $('search-input'), sbox = $('sbox'), nav = $('nav');

const seen = {}, cache = {};
let cur = 'home-view', prev = 'home-view', heroMovie = null, detailMovie = null, listMode = false;

/* ---------- API ---------- */
async function fetchAPI(params) {
  try {
    const res = await fetch(`${API_BASE}?${new URLSearchParams(params).toString()}`);
    const json = await res.json();
    return json.status === 'success' ? json.data : [];
  } catch (err) {
    console.error('Gagal mengambil data:', err);
    return [];
  }
}
async function getList(params) {
  const k = JSON.stringify(params);
  if (cache[k]) return cache[k];
  const d = await fetchAPI(params);
  if (Array.isArray(d) && d.length) cache[k] = d;
  return Array.isArray(d) ? d : [];
}

/* ---------- Daftar Saya (localStorage) ---------- */
const inList = s => store.get('ws_list').some(x => x.slug === s);
function toggleList(m) {
  const l = store.get('ws_list');
  store.set('ws_list', inList(m.slug) ? l.filter(x => x.slug !== m.slug) : [{ slug: m.slug, title: m.title, thumbnail: m.thumbnail }, ...l]);
  syncAdd(); renderUser();
  if (cur === 'results-view' && listMode) showList();
}
function listBtn(b, m) {
  if (!b || !m) return;
  b.innerHTML = `<i class="fa-solid ${inList(m.slug) ? 'fa-check' : 'fa-plus'}"></i> Daftar Saya`;
}
function syncAdd() {
  document.querySelectorAll('.add').forEach(b => {
    const on = inList(b.dataset.s);
    b.classList.toggle('on', on);
    b.firstElementChild.className = 'fa-solid ' + (on ? 'fa-check' : 'fa-plus');
  });
  listBtn($('hero-list'), heroMovie); listBtn($('d-list'), detailMovie);
}

/* ---------- Komponen ---------- */
const metaOf = m => [
  m.rating && `<span><i class="fa-solid fa-star"></i> ${esc(m.rating)}</span>`,
  m.year && `<span>${esc(m.year)}</span>`,
  m.duration && `<span>${esc(m.duration)}</span>`,
  m.genre && `<span>${esc([].concat(m.genre).join(', '))}</span>`
].filter(Boolean).join('');

function card(m, o = {}) {
  seen[m.slug] = seen[m.slug] || { slug: m.slug, title: m.title, thumbnail: m.thumbnail };
  const el = document.createElement('div');
  el.className = 'card';
  const mt = metaOf(m);
  el.innerHTML = (o.top ? `<span class="num">${o.i + 1}</span>` : '') +
    `<div><div class="pw">${m.thumbnail ? `<img src="${esc(m.thumbnail)}" alt="${esc(m.title)}" loading="lazy">` : ''}
      <div class="ov"><span class="play"><i class="fa-solid fa-play"></i></span><b>${esc(m.title)}</b></div>
      <button class="add" data-s="${esc(m.slug)}" aria-label="Daftar Saya"><i class="fa-solid fa-plus"></i></button></div>` +
    (o.top ? '' : `<div class="ct" title="${esc(m.title)}">${esc(m.title)}</div>${mt ? `<div class="cm">${mt}</div>` : ''}`) +
    (o.progress && m.pct ? `<div class="bar"><i style="width:${Math.min(100, +m.pct)}%"></i></div>` : '') + `</div>`;
  el.addEventListener('click', () => openMovieDetail(m.slug));
  el.querySelector('.add').addEventListener('click', e => { e.stopPropagation(); toggleList(m); });
  return el;
}

function rail(title, movies, o = {}) {
  const s = document.createElement('section');
  s.className = 'sec' + (o.cls ? ' ' + o.cls : '');
  s.innerHTML = `<h2>${esc(title)}</h2><div class="rail"></div>`;
  const r = s.querySelector('.rail');
  movies.forEach((m, i) => r.appendChild(card(m, { ...o, i })));
  return s;
}

/* ---------- Beranda ---------- */
async function setHero(m) {
  heroMovie = m;
  heroBanner.style.backgroundImage = m.thumbnail ? `url("${String(m.thumbnail).replace(/["\\\n]/g, '')}")` : '';
  $('hero-title').textContent = m.title;
  $('hero-meta').innerHTML = metaOf(m);
  $('hero-syn').textContent = '';
  $('hero-play').onclick = () => openMovieDetail(m.slug);
  $('hero-list').onclick = () => toggleList(m);
  listBtn($('hero-list'), m);
  const d = await fetchAPI({ slug: m.slug });
  if (heroMovie !== m || !d || Array.isArray(d)) return;
  if (d.synopsis) $('hero-syn').textContent = d.synopsis;
  $('hero-meta').innerHTML = metaOf({ ...m, ...d });
}

function renderUser() {
  const c = store.get('ws_continue'), l = store.get('ws_list');
  $('sec-cont').replaceChildren(...(c.length ? [rail('Lanjutkan Menonton', c, { progress: 1 })] : []));
  $('sec-list').replaceChildren(...(l.length ? [rail('Daftar Saya', l)] : []));
}

async function renderRails() {
  const box = $('rails');
  if (!box.children.length) box.innerHTML = '<p class="empty">Memuat film...</p>';
  const lists = await Promise.all(RAILS.map(r => getList(r.params)));
  const first = lists.find(l => l.length);
  if (!first) {
    box.innerHTML = '<p class="empty">Film belum dapat dimuat. <a href="#" id="retry">Coba lagi</a></p>';
    $('retry').onclick = e => { e.preventDefault(); renderRails(); };
    return;
  }
  if (!heroMovie) setHero(first[0]);
  box.innerHTML = '';
  const ri = RAILS.findIndex(r => r.id === 'rating');
  RAILS.forEach((r, i) => {
    if (lists[i].length) box.appendChild(rail(r.title, lists[i]));
    // Top 10 memakai urutan rating dari API (belum ada data ranking mingguan)
    if (i === 0 && ri > -1 && lists[ri].length)
      box.appendChild(rail('Top 10 Rating Tertinggi', lists[ri].slice(0, 10), { top: 1, cls: 't10' }));
  });
  syncAdd();
}

/* ---------- Navigasi view ---------- */
function updNav() { nav.classList.toggle('solid', scrollY > 40 || cur !== 'home-view'); }
addEventListener('scroll', updNav, { passive: true });

function show(v) {
  cur = v;
  ['home-view', 'results-view', 'detail-view'].forEach(id => $(id).classList.toggle('d-none', id !== v));
  scrollTo(0, 0); updNav();
}
function setActive(k) {
  document.querySelectorAll('[data-go]').forEach(a => a.classList.toggle('on', a.dataset.go === k));
}

function fillGrid(movies, emptyMsg) {
  movieContainer.innerHTML = '';
  if (!movies || !movies.length) {
    movieContainer.innerHTML = `<p class="empty" style="grid-column:1/-1">${emptyMsg}</p>`;
    return;
  }
  movies.forEach(m => movieContainer.appendChild(card(m)));
  syncAdd();
}
function loadingGrid(t) {
  show('results-view'); $('res-title').textContent = t;
  movieContainer.innerHTML = '<p class="empty" style="grid-column:1/-1"><i class="fa-solid fa-spinner fa-spin"></i> Memuat film...</p>';
}
async function showAction(action, title) {
  listMode = false; loadingGrid(title);
  fillGrid(await getList({ action }), 'Film tidak ditemukan.');
}
function showList() {
  listMode = true; show('results-view'); $('res-title').textContent = 'Daftar Saya';
  fillGrid(store.get('ws_list'), 'Daftar masih kosong. Tekan + pada poster untuk menyimpan film.');
}
async function executeSearch(query) {
  listMode = false; loadingGrid(`Hasil untuk "${query}"`);
  fillGrid(await fetchAPI({ action: 'search', query }), 'Film tidak ditemukan.');
}

/* ---------- Detail & Player ---------- */
function showPlayerLoading() { loadingOverlay.classList.remove('d-none'); detailIframe.classList.add('d-none'); }
function hidePlayerLoading() { loadingOverlay.classList.add('d-none'); detailIframe.classList.remove('d-none'); }

detailIframe.addEventListener('load', () => {
  if (detailIframe.src && detailIframe.src !== window.location.href) hidePlayerLoading();
});

function toggleToHomeView() {
  detailIframe.src = '';
  show(prev === 'detail-view' ? 'home-view' : prev);
  if (cur === 'home-view') renderUser();
}
$('back-home-btn').addEventListener('click', toggleToHomeView);

async function openMovieDetail(slug) {
  if (cur !== 'detail-view') prev = cur;
  show('detail-view');
  detailMovie = seen[slug] || { slug };
  listBtn($('d-list'), detailMovie);
  $('d-list').onclick = () => toggleList(detailMovie);
  detailTitle.textContent = 'Memuat judul...';
  $('info-meta').innerHTML = ''; $('info-synopsis').textContent = ''; $('related').innerHTML = '';
  detailServerGrid.innerHTML = '<p class="hint">Mencari server...</p>';
  loadingOverlay.innerHTML = '<i class="fa-solid fa-spinner fa-spin"></i><p>Memuat video</p><span>Mencoba server streaming...</span>';
  detailIframe.src = '';
  showPlayerLoading();

  const detail = await fetchAPI({ slug });

  if (detail && detail.serverPlayer && detail.serverPlayer.length > 0) {
    const mainTitle = detail.title || 'Nonton Film';
    detailTitle.textContent = mainTitle;
    detailMovie = { ...detailMovie, slug, title: mainTitle, thumbnail: detailMovie.thumbnail || detail.thumbnail };
    listBtn($('d-list'), detailMovie);
    $('info-meta').innerHTML = metaOf(detail) + '<span class="hd">HD</span>';
    $('info-synopsis').textContent = detail.synopsis || '';

    detailServerGrid.innerHTML = '';
    detail.serverPlayer.forEach((srv, index) => {
      const btn = document.createElement('button');
      btn.className = 'sb' + (index === 0 ? ' active' : '');
      btn.innerHTML = `<span class="dot"></span>${esc(srv.server)}`;
      btn.addEventListener('click', () => {
        document.querySelectorAll('.sb').forEach(b => b.classList.remove('active'));
        btn.classList.add('active');
        showPlayerLoading();
        detailIframe.src = srv.embed;
      });
      detailServerGrid.appendChild(btn);
    });
    detailIframe.src = detail.serverPlayer[0].embed;

    // Lanjutkan Menonton: simpan film terakhir dibuka (progress menit belum dilacak)
    const k = { slug, title: mainTitle, thumbnail: detailMovie.thumbnail };
    store.set('ws_continue', [k, ...store.get('ws_continue').filter(x => x.slug !== slug)].slice(0, 20));

    // Film terkait: dari daftar API yang tersedia
    const rel = (await getList({ action: 'home' })).filter(m => m.slug !== slug);
    if (rel.length && cur === 'detail-view') { $('related').replaceChildren(rail('Film Lainnya', rel)); syncAdd(); }
  } else {
    detailTitle.textContent = 'Video tidak tersedia';
    detailServerGrid.innerHTML = '<p class="hint">Tidak ada server aktif.</p>';
    loadingOverlay.innerHTML = '<p>Video tidak ditemukan</p><span>Coba film lain atau kembali nanti.</span>';
  }
}

/* ---------- Pencarian ---------- */
function openSearch() {
  sbox.classList.add('open'); document.body.classList.add('searching');
  setTimeout(() => searchInput.focus(), 50);
}
function closeSearch() {
  if (searchInput.value.trim()) return;
  sbox.classList.remove('open'); document.body.classList.remove('searching');
}
$('search-btn').addEventListener('click', () => {
  if (sbox.classList.contains('open') && searchInput.value.trim()) triggerSearch();
  else if (sbox.classList.contains('open')) { searchInput.blur(); closeSearch(); }
  else openSearch();
});
function triggerSearch() {
  const q = searchInput.value.trim();
  if (!q) return;
  if (cur === 'detail-view') detailIframe.src = '';
  setActive(''); executeSearch(q); searchInput.blur();
}
searchInput.addEventListener('keydown', e => {
  if (e.key === 'Enter') triggerSearch();
  if (e.key === 'Escape') { searchInput.value = ''; searchInput.blur(); }
});
searchInput.addEventListener('blur', closeSearch);

/* ---------- Menu ---------- */
document.querySelectorAll('[data-go]').forEach(a => a.addEventListener('click', e => {
  e.preventDefault();
  const k = a.dataset.go;
  if (cur === 'detail-view') detailIframe.src = '';
  if (k === 'search') { openSearch(); return; }
  searchInput.value = ''; closeSearch(); setActive(k);
  if (k === 'home') { listMode = false; show('home-view'); renderUser(); renderRails(); }
  else if (k === 'new') showAction('home', 'Terbaru Ditambahkan');
  else if (k === 'rating') showAction('rating', 'Rating Tertinggi');
  else if (k === 'list') showList();
}));
